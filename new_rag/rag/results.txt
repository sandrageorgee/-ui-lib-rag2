```tsx
import React, { useState, useEffect } from 'react';
import { Button, ButtonProps } from './Button';

interface TreeNode {
  id: number;
  name: string;
  children?: TreeNode[];
}

const DashboardPage: React.FC = () => {
  const [treeData, setTreeData] = useState<TreeNode[]>([]);
  const [loading, setLoading] = useState<boolean>(true);

  // Mock initial tree data
  const initialTreeData: TreeNode[] = [
    {
      id: 1,
      name: 'Dashboard',
      children: [
        { id: 2, name: 'Analytics' },
        { id: 3, name: 'Reports' },
        {
          id: 4,
          name: 'Settings',
          children: [
            { id: 5, name: 'General' },
            { id: 6, name: 'Privacy' },
          ],
        },
      ],
    },
    {
      id: 7,
      name: 'Users',
      children: [
        { id: 8, name: 'Active Users' },
        { id: 9, name: 'Pending Users' },
      ],
    },
  ];

  // Simulate data loading
  useEffect(() => {
    const timer = setTimeout(() => {
      setTreeData(initialTreeData);
      setLoading(false);
    }, 500);
    return () => clearTimeout(timer);
  }, []);

  // Refresh handler
  const handleRefresh = () => {
    setLoading(true);
    const timer = setTimeout(() => {
      setTreeData([...initialTreeData]); // Create new reference to trigger re-render
      setLoading(false);
    }, 500);
    return () => clearTimeout(timer);
  };

  // Recursive tree renderer
  const renderTree = (nodes: TreeNode[]) => {
    return (
      <ul style={{ listStyle: 'none', paddingLeft: '20px', margin: 0 }}>
        {nodes.map((node) => (
          <li key={node.id} style={{ margin: '8px 0' }}>
            <div style={{ display: 'flex', alignItems: 'center' }}>
              <span style={{ marginRight: '8px' }}>
                {node.children ? '📁' : '📄'}
              </span>
              <span>{node.name}</span>
            </div>
            {node.children && renderTree(node.children)}
          </li>
        ))}
      </ul>
    );
  };

  return (
    <div style={{ padding: '24px', fontFamily: 'Arial, sans-serif' }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '24px' }}>
        <h1 style={{ margin: 0 }}>Dashboard</h1>
        <Button
          onClick={handleRefresh}
          labelType="refresh"
          variant="primary"
          disabled={loading}
          style={{ minWidth: '120px' }}
        >
          {loading ? 'Refreshing...' : 'Refresh'}
        </Button>
      </div>

      <div style={{ 
        background: '#f5f5f5', 
        borderRadius: '8px', 
        padding: '20px',
        minHeight: '400px'
      }}>
        <h2 style={{ marginTop: 0, marginBottom: '16px' }}>Navigation Tree</h2>
        
        {loading ? (
          <div style={{ textAlign: 'center', padding: '40px' }}>
            Loading tree data...
          </div>
        ) : (
          <div style={{ 
            background: 'white', 
            borderRadius: '4px', 
            padding: '20px',
            border: '1px solid #ddd'
          }}>
            {treeData.length === 0 ? (
              <div style={{ textAlign: 'center', padding: '40px' }}>
                No tree data available
              </div>
            ) : (
              renderTree(treeData)
            )}
          </div>
        )}

        <div style={{ marginTop: '20px', fontSize: '14px', color: '#666' }}>
          <p>Total Nodes: {countNodes(treeData)}</p>
          <p>Last refreshed: {new Date().toLocaleTimeString()}</p>
        </div>
      </div>
    </div>
  );
};

// Helper function to count all nodes in tree
const countNodes = (nodes: TreeNode[]): number => {
  return nodes.reduce((count, node) => {
    return count + 1 + (node.children ? countNodes(node.children) : 0);
  }, 0);
};

export default DashboardPage;
```
